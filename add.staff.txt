  document.addEventListener('DOMContentLoaded', function() {

    // Name concatenation function
    function updateFullName() {
      const salutation = document.getElementById('salutation').value || '';
      const firstName = document.getElementById('first_name').value || '';
      const middleName = document.getElementById('middle_name').value || '';
      const lastName = document.getElementById('last_name').value || '';
      
      let fullName = '';
      if (salutation) fullName += salutation + ' ';
      if (firstName) fullName += firstName + ' ';
      if (middleName) fullName += middleName + ' ';
      if (lastName) fullName += lastName;
      
      fullName = fullName.trim();
      
      // Update hidden field and display
      document.getElementById('full_name').value = fullName;
      document.getElementById('fullNameDisplay').textContent = fullName || '[Full name will appear here]';
    }
    
    // Retirement calculation function
    function calculateRetirement() {
      const dobInput = document.getElementById('date_of_birth').value;
      const disability = document.getElementById('disability').value;
      
      if (!dobInput || !disability) {
        document.getElementById('calculated_age').value = '-';
        document.getElementById('retirement_date').value = '-';
        document.getElementById('retirementAge').textContent = '60';
        return;
      }
      
      // Parse DD/MM/YYYY
      const parts = dobInput.split('/');
      if (parts.length !== 3) return;
      
      const day = parseInt(parts[0]);
      const month = parseInt(parts[1]) - 1; // JS months are 0-indexed
      const year = parseInt(parts[2]);
      
      if (isNaN(day) || isNaN(month) || isNaN(year)) return;
      
      const birthDate = new Date(year, month, day);
      const today = new Date();
      
      // Calculate age
      let age = today.getFullYear() - birthDate.getFullYear();
      const monthDiff = today.getMonth() - birthDate.getMonth();
      if (monthDiff < 0 || (monthDiff === 0 && today.getDate() < birthDate.getDate())) {
        age--;
      }
      
      // Set retirement age (60 for non-PWD, 65 for PWD)
      const retirementAge = disability === 'Yes' ? 65 : 60;
      document.getElementById('retirementAge').textContent = retirementAge;
      
      // Calculate retirement date: birth date + retirement age years - 1 day
      const retirementDate = new Date(birthDate);
      retirementDate.setFullYear(retirementDate.getFullYear() + retirementAge);
      retirementDate.setDate(retirementDate.getDate() - 1);
      
      // Format dates
      const formatDate = (date) => {
        const d = date.getDate().toString().padStart(2, '0');
        const m = (date.getMonth() + 1).toString().padStart(2, '0');
        const y = date.getFullYear();
        return `${d}/${m}/${y}`;
      };
      
      // Update fields
      document.getElementById('calculated_age').value = age;
      document.getElementById('retirement_date').value = formatDate(retirementDate);
    }
    
    // Event listeners for name fields
    ['salutation', 'first_name', 'middle_name', 'last_name'].forEach(id => {
      document.getElementById(id).addEventListener('input', updateFullName);
      document.getElementById(id).addEventListener('change', updateFullName);
    });
    
    // Event listeners for retirement calculation
    ['date_of_birth', 'disability'].forEach(id => {
      document.getElementById(id).addEventListener('input', calculateRetirement);
      document.getElementById(id).addEventListener('change', calculateRetirement);
    });
    
    // Form submission
    const employeeForm = document.getElementById('employeeForm');
    if (employeeForm) {
      employeeForm.addEventListener('submit', function(e) {
        e.preventDefault();
        
        // Validate retirement date is calculated
        if (document.getElementById('retirement_date').value === '-') {
          alert('Please enter Date of Birth and Disability status to calculate retirement date.');
          document.getElementById('date_of_birth').focus();
          return;
        }
        
        // Prepare form data for submission
        const formData = new FormData(this);
        
        // Convert gender for backend (already M/F)
        // Convert status for backend: Active → "0 - Active", Inactive → "1 - Inactive"
        const status = formData.get('status');
        if (status === 'Active') {
          formData.set('status', '0 - Active');
        } else if (status === 'Inactive') {
          formData.set('status', '1 - Inactive');
        }
        
        // Convert employment terms to employment_status
        formData.set('employment_status', formData.get('employment_terms'));
        
        // Add calculated age to form data
        formData.set('age', document.getElementById('calculated_age').value);
        
        // For now, just show what would be sent
        console.log('Form data to submit:', Object.fromEntries(formData));
        
        // In production: Send to server
        // fetch('/employees/add', {
        //   method: 'POST',
        //   body: formData
        // })
        // .then(response => response.json())
        // .then(data => {
        //   alert('Employee added successfully!');
        //   this.reset();
        //   updateFullName();
        //   calculateRetirement();
        // })
        // .catch(error => {
        //   console.error('Error:', error);
        //   alert('Error adding employee.');
        // });
        
        alert('Employee form validated and ready for submission!');
        this.reset();
        updateFullName();
        calculateRetirement();
      });
    }
    
    // Form validation
    (function() {
      'use strict';
      var forms = document.querySelectorAll('.needs-validation');
      Array.prototype.slice.call(forms).forEach(function(form) {
        form.addEventListener('submit', function(event) {
          if (!form.checkValidity()) {
            event.preventDefault();
            event.stopPropagation();
          }
          form.classList.add('was-validated');
        }, false);
      });
    })();
    
    // Switch between forms
    window.switchToSingleForm = function() {
      document.getElementById('singleEmployeeForm').style.display = 'block';
      document.getElementById('bulkUploadForm').style.display = 'none';
      document.querySelectorAll('#table-simple-pagination-replace-element button').forEach(btn => {
        btn.classList.remove('active');
      });
      document.querySelector('#table-simple-pagination-replace-element button:first-child').classList.add('active');
    };

      document.getElementById('singleEmployeeForm').style.display = 'none';
      document.getElementById('bulkUploadForm').style.display = 'block';

        btn.classList.remove('active');
      });
      document.querySelector('#table-simple-pagination-replace-element button:last-child').classList.add('active');
    };
    
    // Initialize calculations
    updateFullName();
    calculateRetirement();
  });
